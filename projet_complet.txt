"""
MenuFlash Mini — point d'entrée Flask
Menu affiché par QR code + commande en ligne (sans paiement en ligne).
+ Page de présentation (landing page) publique + inscription en
  libre-service, avec le même langage visuel appliqué à l'ensemble du
  back-office.
+ Notifications cuisine instantanées (Server-Sent Events).
+ Annulation de commande par le client tant qu'elle est "nouvelle".
+ Thème de couleur par restaurant, appliqué partout et propagé en direct.
+ Panel super-admin, QR codes par table, glisser-déposer, panneau "Mes
  commandes" côté client, ajout à une commande existante, statistiques
  de ventes, comptes d'équipe à rôles (owner / staff).
+ Avis clients par plat, interface multi-langue FR/EN, temps de
  préparation estimé, mode "rush" (pause de catégorie), historique des
  modifications du menu, modification en place des plats/catégories.
+ SUGGESTIONS DE VENTE ADDITIONNELLE : un plat peut en "suggérer" un autre.
+ MODE MULTI-RESTAURANT : un même compte "owner" peut créer et gérer
  plusieurs établissements, et basculer entre eux depuis un sélecteur.
+ IDENTITÉ VISUELLE DU RESTAURANT (marque blanche partielle) : la barre
  de navigation du back-office affiche le NOM et le LOGO du restaurant
  actif, à la place de la marque "MenuFlash Mini" — seul le panel
  super-admin et les pages non rattachées à un restaurant (connexion,
  inscription, mot de passe oublié) gardent la marque de la plateforme.
+ NAVIGATION RÉORGANISÉE EN MENUS DÉROULANTS : les nombreuses pages du
  back-office sont regroupées en 3 menus ("Boutique", "Gestion",
  "Compte") + 2 liens directs ("Menu", "Commandes"), via un unique
  template partiel (templates/_admin_navbar.html) inclus par toutes les
  pages — une seule source de vérité pour éviter les divergences.

SÉCURITÉ :
  - Protection CSRF (Flask-WTF) sur tous les formulaires et appels AJAX.
  - Limitation du débit (Flask-Limiter) sur la connexion et l'inscription.
  - En-têtes de sécurité HTTP (CSP, anti-clickjacking, anti-sniffing).
  - Cookies de session HttpOnly + SameSite=Lax (+ Secure en production).
  - HSTS + redirection forcée vers HTTPS (via FORCE_HTTPS=true en prod).
  - Verrouillage de compte après 5 échecs de connexion (15 minutes).
  - Invalidation de TOUTES les sessions actives ailleurs dès qu'un mot de
    passe est changé ou réinitialisé (tampon de sécurité par utilisateur).
  - Vérification d'adresse email à l'inscription (lien à usage unique).
  - Réinitialisation de mot de passe sécurisée (jeton signé, à durée de
    vie limitée, à usage unique).
  - Journal d'audit des événements de sécurité.
  - Validation et troncature serveur de toutes les entrées utilisateur.
  - Mots de passe hachés (PBKDF2 via Werkzeug), jamais stockés en clair.

Lancement local (Windows / PowerShell) :
    $env:FLASK_SECRET_KEY = "change-moi"
    python app.py

@app.before_request
def force_https_redirect():
    if not FORCE_HTTPS:
        return None
    proto = request.headers.get("X-Forwarded-Proto", "https" if request.is_secure else "http")
    if proto != "https":
        url = request.url.replace("http://", "https://", 1)
        return redirect(url, code=301)
    return None


@app.after_request
def set_security_headers(response):
    response.headers["X-Content-Type-Options"] = "nosniff"
    response.headers["X-Frame-Options"] = "DENY"
    response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
    response.headers["Permissions-Policy"] = "geolocation=(), microphone=(), camera=()"
    response.headers["Content-Security-Policy"] = (
        "default-src 'self'; "
        "script-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net; "
        "style-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net https://fonts.googleapis.com; "
        "font-src 'self' https://fonts.gstatic.com https://cdn.jsdelivr.net; "
        "img-src 'self' data: blob:; "
        "connect-src 'self';"
    )
    is_https = request.is_secure or request.headers.get("X-Forwarded-Proto") == "https"
    if is_https or FORCE_HTTPS:
        response.headers["Strict-Transport-Security"] = "max-age=31536000; includeSubDomains"
    return response


@login_manager.user_loader
def load_user(composite_id):
    try:
        uid_str, stamp = composite_id.split(":", 1)
        uid = int(uid_str)
    except (ValueError, AttributeError):
        return None
    user = User.query.get(uid)
    if not user or user.security_stamp != stamp:
        return None
    return user

def get_owner_restaurants(user):
    if not user or not getattr(user, "is_owner", False) or getattr(user, "is_super_admin", False):
        return []
    ids = {user.restaurant_id}
    extra = RestaurantAccess.query.filter_by(user_id=user.id).all()
    ids.update(a.restaurant_id for a in extra)
    return Restaurant.query.filter(Restaurant.id.in_(ids)).order_by(Restaurant.created_at.asc()).all()


def active_restaurant():
    if not current_user.is_authenticated:
        return None
    if current_user.is_super_admin or not current_user.is_owner:
        return current_user.restaurant

    rid = session.get("active_restaurant_id")
    if rid:
        allowed_ids = {current_user.restaurant_id} | {
            a.restaurant_id for a in RestaurantAccess.query.filter_by(user_id=current_user.id).all()
        }
        if rid in allowed_ids:
            r = Restaurant.query.get(rid)
            if r:
                return r
    return current_user.restaurant


def active_restaurant_id():
    r = active_restaurant()
    return r.id if r else None

def slugify(value):
    value = unicodedata.normalize("NFKD", value).encode("ascii", "ignore").decode("ascii")
    value = value.lower().strip()
    value = re.sub(r"[^a-z0-9]+", "-", value)
    value = re.sub(r"-+", "-", value).strip("-")
    return value[:60] or "restaurant"


def slug_is_available(slug):
    if not slug or slug in RESERVED_SLUGS:
        return False
    return Restaurant.query.filter_by(slug=slug).first() is None


def safe_int(raw_value, default=None, min_value=None, max_value=None):
    try:
        value = int(raw_value)
    except (TypeError, ValueError):
        return default
    if min_value is not None:
        value = max(value, min_value)
    if max_value is not None:
        value = min(value, max_value)
    return value


def get_client_ip():
    forwarded = request.headers.get("X-Forwarded-For", "")
    if forwarded:
        return forwarded.split(",")[0].strip()[:64]
    return (request.remote_addr or "unknown")[:64]


def log_audit(action, success=True, restaurant_id=None, user_email=None, details=None):
    try:
        db.session.add(AuditLog(
            restaurant_id=restaurant_id,
            user_email=user_email,
            action=action,
            success=success,
            ip_address=get_client_ip(),
            details=(details or "")[:300],
        ))
        db.session.commit()
    except Exception:
        db.session.rollback()
        logger.exception("Échec d'écriture du journal d'audit (action=%s)", action)


def _token_serializer():
    return URLSafeTimedSerializer(app.config["SECRET_KEY"])


def generate_security_token(user, salt):
    return _token_serializer().dumps({"uid": user.id, "stamp": user.security_stamp}, salt=salt)


def verify_security_token(token, salt, max_age):
    try:
        data = _token_serializer().loads(token, salt=salt, max_age=max_age)
    except (BadSignature, SignatureExpired):
        return None
    user = User.query.get(data.get("uid"))
    if not user or user.security_stamp != data.get("stamp"):
        return None
    return user

@app.route("/inscription", methods=["GET", "POST"])
@limiter.limit("8 per hour", methods=["POST"])
def signup():
    if request.method == "GET":
        return render_template("signup.html")

    restaurant_name = request.form.get("restaurant_name", "").strip()[:120]
    slug_input = request.form.get("slug", "").strip().lower()[:80]
    owner_email = request.form.get("email", "").strip().lower()[:120]
    owner_password = request.form.get("password", "")
    phone = request.form.get("phone", "").strip()[:30]

    errors = []
    if not restaurant_name:
        errors.append("Le nom du restaurant est obligatoire.")
    if not owner_email or "@" not in owner_email:
        errors.append("Une adresse email valide est requise.")
    if len(owner_password) < MIN_PASSWORD_LENGTH:
        errors.append(f"Le mot de passe doit contenir au moins {MIN_PASSWORD_LENGTH} caractères.")

    slug = slugify(slug_input or restaurant_name)
    if not slug_is_available(slug):
        base_slug = slug
        suffix = 2
        while not slug_is_available(f"{base_slug}-{suffix}") and suffix < 50:
            suffix += 1
        slug = f"{base_slug}-{suffix}"

    if User.query.filter_by(email=owner_email).first():
        errors.append("Un compte existe déjà avec cet email.")

    if errors:
        for message in errors:
            flash(message, "error")
        log_audit("signup_failed", success=False, user_email=owner_email,
                   details="; ".join(errors)[:300])
        return render_template(
            "signup.html",
            form_values={"restaurant_name": restaurant_name, "slug": slug_input,
                         "email": owner_email, "phone": phone},
        ), 400

    restaurant = Restaurant(
        name=restaurant_name,
        slug=slug,
        phone=phone or None,
        subscription_plan="trial",
        subscription_active_until=datetime.utcnow() + timedelta(days=TRIAL_DAYS),
    )
    db.session.add(restaurant)
    db.session.commit()

    owner = User(restaurant_id=restaurant.id, email=owner_email, role="owner")
    owner.set_password(owner_password)
    db.session.add(owner)
    db.session.commit()

    send_verification_email(owner)
    log_audit("signup", restaurant_id=restaurant.id, user_email=owner_email,
               details=f"Restaurant « {restaurant_name} » créé (slug={slug}).")

    login_user(owner)
    flash(
        f"Bienvenue sur MenuFlash Mini ! Votre essai gratuit de {TRIAL_DAYS} jours a commencé. "
        f"Votre menu est accessible à l'adresse /m/{slug}. Pensez à confirmer votre email "
        f"(lien envoyé à {owner_email}), et à ajouter le logo de votre restaurant depuis "
        f"la page Apparence.",
        "success",
    )
    return redirect(url_for("admin_dashboard"))

@app.route("/admin/restaurants/new", methods=["POST"])
@login_required
def admin_restaurants_new():
    if not current_user.is_owner or current_user.is_super_admin:
        abort(403)

    name = request.form.get("name", "").strip()[:120]
    slug_input = request.form.get("slug", "").strip().lower()[:80]
    if not name:
        flash("Le nom du restaurant est obligatoire.", "error")
        return redirect(url_for("admin_dashboard"))

    slug = slugify(slug_input or name)
    if not slug_is_available(slug):
        base_slug = slug
        suffix = 2
        while not slug_is_available(f"{base_slug}-{suffix}") and suffix < 50:
            suffix += 1
        slug = f"{base_slug}-{suffix}"

    restaurant = Restaurant(
        name=name, slug=slug,
        subscription_plan="trial",
        subscription_active_until=datetime.utcnow() + timedelta(days=TRIAL_DAYS),
    )
    db.session.add(restaurant)
    db.session.commit()

    db.session.add(RestaurantAccess(user_id=current_user.id, restaurant_id=restaurant.id))
    db.session.commit()

    log_audit("additional_restaurant_created", restaurant_id=restaurant.id,
               user_email=current_user.email,
               details=f"« {name} » créé comme restaurant additionnel (slug={slug}).")

    session["active_restaurant_id"] = restaurant.id
    flash(
        f"Restaurant « {name} » créé avec un essai gratuit de {TRIAL_DAYS} jours. "
        f"Vous pouvez maintenant y ajouter votre menu.",
        "success",
    )
    return redirect(url_for("admin_dashboard"))


@app.route("/admin/restaurants/switch", methods=["POST"])
@login_required
def admin_restaurants_switch():
    if not current_user.is_owner or current_user.is_super_admin:
        abort(403)

    target_id = safe_int(request.form.get("restaurant_id"))
    allowed_ids = {current_user.restaurant_id} | {
        a.restaurant_id for a in RestaurantAccess.query.filter_by(user_id=current_user.id).all()
    }
    if target_id not in allowed_ids:
        flash("Vous n'avez pas accès à ce restaurant.", "error")
        return redirect(url_for("admin_dashboard"))

    session["active_restaurant_id"] = target_id
    r = Restaurant.query.get(target_id)
    log_audit("restaurant_switched", restaurant_id=target_id, user_email=current_user.email,
               details=f"Bascule vers « {r.name if r else target_id} ».")
    return redirect(url_for("admin_dashboard"))

@app.route("/verifier-email/<token>")
def verify_email(token):
    user = verify_security_token(token, salt="email-verify", max_age=EMAIL_VERIFY_MAX_AGE)
    if not user:
        flash("Ce lien de vérification est invalide ou a expiré. Demandez-en un nouveau depuis votre tableau de bord.", "error")
        return redirect(url_for("admin_login"))

    if not user.email_verified:
        user.email_verified = True
        user.email_verified_at = datetime.utcnow()
        db.session.commit()
        log_audit("email_verified", restaurant_id=user.restaurant_id, user_email=user.email)
        flash("Votre adresse email a été confirmée. Merci !", "success")
    else:
        flash("Votre adresse email était déjà confirmée.", "success")

    if current_user.is_authenticated:
        return redirect(url_for("admin_orders") if not current_user.is_owner else url_for("admin_dashboard"))
    return redirect(url_for("admin_login"))

@app.route("/admin/reinitialiser/<token>", methods=["GET", "POST"])
@limiter.limit("10 per hour", methods=["POST"])
def reset_password(token):
    user = verify_security_token(token, salt="pwd-reset", max_age=PASSWORD_RESET_MAX_AGE)
    if not user:
        flash("Ce lien de réinitialisation est invalide, a expiré, ou a déjà été utilisé.", "error")
        return redirect(url_for("forgot_password"))

    if request.method == "GET":
        return render_template("reset_password.html", token=token)

    new_password = request.form.get("password", "")

def allowed_file(filename):
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXT


def save_compressed_image(file_storage, restaurant_slug):
    """
    Recompresse une photo de plat (toujours aplatie en RGB — un fond
    blanc derrière un plat n'a aucun inconvénient visuel, contrairement
    à un logo). Voir save_logo_image() pour la variante qui préserve la
    transparence.
    """
    filename = secure_filename(file_storage.filename)
    if not allowed_file(filename):
        return None

    try:
        img = Image.open(file_storage.stream)
        img.verify()
        file_storage.stream.seek(0)
        img = Image.open(file_storage.stream).convert("RGB")
    except (UnidentifiedImageError, OSError, ValueError):
        return None

    img.thumbnail((MAX_IMAGE_SIDE, MAX_IMAGE_SIDE))

    out_name = f"{restaurant_slug}_{int(datetime.utcnow().timestamp())}.webp"
    out_path = os.path.join(UPLOAD_DIR, out_name)
    img.save(out_path, "WEBP", quality=70)
    return f"/static/uploads/{out_name}"


def save_logo_image(file_storage, restaurant_slug):
    """
    Recompresse un logo de restaurant. Contrairement aux photos de plats,
    un logo a très souvent un fond transparent (PNG) qu'il est important
    de préserver — sinon il apparaît avec un carré blanc disgracieux sur
    la barre de navigation sombre. Si l'image source a un canal alpha
    (RGBA, LA, ou palette avec transparence), le fichier de sortie reste
    un PNG avec transparence ; sinon on utilise WEBP (plus léger) comme
    pour les photos de plats.
    """
    filename = secure_filename(file_storage.filename)
    if not allowed_file(filename):
        return None

    try:
        img = Image.open(file_storage.stream)
        img.verify()
        file_storage.stream.seek(0)
        img = Image.open(file_storage.stream)
        has_alpha = img.mode in ("RGBA", "LA") or (img.mode == "P" and "transparency" in img.info)
        img = img.convert("RGBA") if has_alpha else img.convert("RGB")
    except (UnidentifiedImageError, OSError, ValueError):
        return None

    img.thumbnail((MAX_LOGO_SIDE, MAX_LOGO_SIDE))

    timestamp = int(datetime.utcnow().timestamp())
    if has_alpha:
        out_name = f"{restaurant_slug}-logo_{timestamp}.png"
        out_path = os.path.join(UPLOAD_DIR, out_name)
        img.save(out_path, "PNG", optimize=True)
    else:
        out_name = f"{restaurant_slug}-logo_{timestamp}.webp"
        out_path = os.path.join(UPLOAD_DIR, out_name)
        img.save(out_path, "WEBP", quality=85)
    return f"/static/uploads/{out_name}"

        "editable": order.status == EDITABLE_STATUS,

def _order_feed_dict(order, now=None):
    now = now or datetime.utcnow()
    return {
        "id": order.id,
        "status": order.status,
        "table_label": order.table_label or "Sans table",
        "notes": order.notes,
        "total": order.total,
        "elapsed_minutes": max(int((now - order.created_at).total_seconds() // 60), 0),
        "items": [{"name": it.item_name, "quantity": it.quantity} for it in order.items],
    }


def _parse_requested_items(restaurant):
    available_items = MenuItem.query.filter_by(
        restaurant_id=restaurant.id, is_available=True).all()
    items_by_id = {str(mi.id): mi for mi in available_items}

    items_to_order = []
    for key, value in request.form.items():
        if key.startswith("qty_"):
            item_id = key[4:]
            qty = safe_int(value, default=0)
            if qty and qty > 0 and item_id in items_by_id:
                items_to_order.append((items_by_id[item_id], min(qty, 50)))
    return items_to_order


def _dish_rating_map(restaurant_id):
    rows = (
        db.session.query(DishReview.menu_item_id,
                          db.func.avg(DishReview.rating),
                          db.func.count(DishReview.id))
        .filter(DishReview.restaurant_id == restaurant_id, DishReview.menu_item_id.isnot(None))
        .group_by(DishReview.menu_item_id)
        .all()
    )
    return {mid: {"avg": round(float(avg), 1), "count": int(count)} for mid, avg, count in rows}


def _build_item_suggestions_map(restaurant_id):
    rows = (
        db.session.query(MenuItemSuggestion.menu_item_id, MenuItem)
        .join(MenuItem, MenuItem.id == MenuItemSuggestion.suggested_item_id)
        .filter(
            MenuItemSuggestion.restaurant_id == restaurant_id,
            MenuItem.is_available.is_(True),
            MenuItemSuggestion.menu_item_id != MenuItemSuggestion.suggested_item_id,
        )
        .all()
    )
    mapping = {}
    for source_id, suggested_item in rows:
        bucket = mapping.setdefault(source_id, [])
        if len(bucket) < MAX_UPSELL_SUGGESTIONS_PER_ITEM:
            bucket.append({
                "id": suggested_item.id,
                "name": suggested_item.name,
                "price": suggested_item.price,
            })
    return mapping

@app.route("/m/<slug>")
def public_menu(slug):
    restaurant = tenant_or_404(slug)

    if not restaurant.is_subscription_valid:
        return render_template("subscription_expired.html", restaurant=restaurant)

    table_label = request.args.get("t")

    qr_id = request.args.get("qr", type=int)
    if qr_id:
        qr = QRCode.query.filter_by(id=qr_id, restaurant_id=restaurant.id).first()
        if qr:
            qr.scan_count = (qr.scan_count or 0) + 1
            db.session.commit()
            table_label = qr.label

    categories = (MenuCategory.query
                  .filter_by(restaurant_id=restaurant.id, is_paused=False)
                  .order_by(MenuCategory.position)
                  .all())

    ratings = _dish_rating_map(restaurant.id)
    item_suggestions = _build_item_suggestions_map(restaurant.id)

    return render_template(
        "public_menu.html",
        restaurant=restaurant,
        categories=categories,
        table_label=table_label,
        ratings=ratings,
        item_suggestions=item_suggestions,
        translations=TRANSLATIONS,
    )


def _build_qr_target_url(restaurant, qr=None, table_label=None):

        params.append(f"qr={qr.id}")
    elif table_label:
        params.append(f"t={table_label}")
    if params:
        target_url += "?" + "&".join(params)
    return target_url

@app.route("/m/<slug>/theme-stream")
def public_theme_stream(slug):
    restaurant = tenant_or_404(slug)
    q = realtime.subscribe(restaurant.id)

    def event_stream():
        try:
            yield "retry: 3000\n\n"
            while True:
                try:
                    message = q.get(timeout=15)
                    try:
                        parsed = json.loads(message)
                    except Exception:
                        continue
                    if parsed.get("type") == "theme_changed":
                        yield realtime.format_sse(message)
                except Exception:
                    yield ": ping\n\n"
        finally:
            realtime.unsubscribe(restaurant.id, q)

    return Response(event_stream(), mimetype="text/event-stream", headers={
        "Cache-Control": "no-cache",
        "X-Accel-Buffering": "no",
    })

@app.route("/m/<slug>/commande", methods=["POST"])
def place_order(slug):
    restaurant = tenant_or_404(slug)
    wants_json = request.headers.get("X-Requested-With") == "fetch"

    if not restaurant.is_subscription_valid:
        abort(404)

    table_label = (request.form.get("table_label") or "").strip()[:50] or None
    qr_id = request.form.get("qr_id", type=int)
    items_to_order = _parse_requested_items(restaurant)

    if not items_to_order:
        message = "Veuillez sélectionner au moins un plat avant d'envoyer la commande."
        if wants_json:
            return jsonify({"ok": False, "error": message}), 400
        flash(message, "error")
        params = {}
        if table_label:
            params["t"] = table_label
        if qr_id:
            params["qr"] = qr_id
        return redirect(url_for("public_menu", slug=slug, **params))

    order = Order(
        restaurant_id=restaurant.id,
        qr_code_id=qr_id,
        table_label=table_label,
        notes=(request.form.get("notes") or "").strip()[:300],
    )
    db.session.add(order)
    db.session.flush()

    for menu_item, qty in items_to_order:
        db.session.add(OrderItem(
            order_id=order.id,
            menu_item_id=menu_item.id,
            item_name=menu_item.name,
            item_price=menu_item.price,
            quantity=qty,
            prep_minutes=menu_item.prep_minutes or DEFAULT_PREP_MINUTES,
        ))
    db.session.commit()

    realtime.publish(restaurant.id, "new_order", _order_feed_dict(order))

    if wants_json:
        data = _order_to_dict(order)
        data["ok"] = True
        data["detail_url"] = url_for("order_status", slug=slug, order_id=order.id)
        return jsonify(data)

    return redirect(url_for("order_status", slug=slug, order_id=order.id))

@app.route("/m/<slug>/commande/<int:order_id>/avis", methods=["POST"])
def submit_review(slug, order_id):
    restaurant = tenant_or_404(slug)
    order = Order.query.filter_by(id=order_id, restaurant_id=restaurant.id).first_or_404()

    if order.status != "servie":
        return jsonify({"ok": False, "error": "Cette commande n'est pas encore marquée comme servie."}), 409
    if order.reviewed:
        return jsonify({"ok": False, "error": "Vous avez déjà donné votre avis pour cette commande."}), 409

    any_rating = False
    for item in order.items:
        raw = request.form.get(f"rating_{item.id}")
        if not raw:
            continue
        rating = safe_int(raw)
        if rating is not None and 1 <= rating <= 5:
            db.session.add(DishReview(
                restaurant_id=restaurant.id,
                order_id=order.id,
                menu_item_id=item.menu_item_id,
                item_name=item.item_name,
                rating=rating,
            ))
            any_rating = True

    if not any_rating:
        return jsonify({"ok": False, "error": "Veuillez noter au moins un plat."}), 400

    order.comment = (request.form.get("comment") or "").strip()[:500] or None
    order.reviewed = True
    db.session.commit()

    return jsonify({"ok": True})

@app.route("/admin/login", methods=["GET", "POST"])
@limiter.limit("15 per minute;50 per hour", methods=["POST"])
def admin_login():
    if request.method == "POST":
        email = request.form.get("email", "").strip().lower()[:120]
        password = request.form.get("password", "")
        user = User.query.filter_by(email=email).first()

        if user and user.is_locked:
            minutes = user.lock_minutes_remaining
            flash(f"Compte temporairement verrouillé suite à plusieurs échecs de connexion. "
                  f"Réessayez dans environ {minutes} minute(s).", "error")
            log_audit("login_blocked_locked", success=False, restaurant_id=user.restaurant_id,
                       user_email=email, details=f"verrouille encore {minutes} min")
            return render_template("admin_login.html")

        if user and user.check_password(password):
            user.register_successful_login()
            db.session.commit()
            login_user(user)
            session.pop("active_restaurant_id", None)
            log_audit("login_success", restaurant_id=user.restaurant_id, user_email=email)
            if user.is_super_admin:
                return redirect(url_for("super_admin_dashboard"))
            if not user.is_owner:
                return redirect(url_for("admin_orders"))
            return redirect(url_for("admin_dashboard"))

        if user:
            user.register_failed_login()
            db.session.commit()
            just_locked = user.is_locked
            log_audit("login_failed", success=False, restaurant_id=user.restaurant_id,

@app.route("/admin/items/<int:item_id>/edit", methods=["POST"])
@owner_required
def edit_item(item_id):
    restaurant = active_restaurant()
    item = MenuItem.query.filter_by(
        id=item_id, restaurant_id=restaurant.id).first_or_404()

    new_name = request.form.get("name", "").strip()[:120]
    new_price = safe_int(request.form.get("price"), min_value=0, max_value=10_000_000)
    if not new_name or new_price is None:
        flash("Le nom et le prix sont obligatoires.", "error")
        return redirect(url_for("admin_dashboard"))

    new_prep = safe_int(request.form.get("prep_minutes"), default=DEFAULT_PREP_MINUTES,
                         min_value=1, max_value=180)
    new_description = request.form.get("description", "").strip()[:300]

    changes = []
    if new_name != item.name:
        changes.append(f"nom « {item.name} » → « {new_name} »")
    if new_price != item.price:
        changes.append(f"prix {item.price} → {new_price} {restaurant.currency}")
    if new_prep != item.prep_minutes:
        changes.append(f"préparation {item.prep_minutes} → {new_prep} min")
    if new_description != (item.description or ""):
        changes.append("description modifiée")

    file = request.files.get("photo")
    if file and file.filename:
        photo_url = save_compressed_image(file, restaurant.slug)
        if photo_url:
            item.photo_url = photo_url
            changes.append("photo mise à jour")

    item.name = new_name
    item.description = new_description
    item.price = new_price
    item.prep_minutes = new_prep

    submitted_ids = set()
    for raw in request.form.getlist("suggested_item_ids"):
        sid = safe_int(raw)
        if sid is not None and sid != item.id:
            submitted_ids.add(sid)

    if submitted_ids:
        valid_ids = {mi.id for mi in MenuItem.query.filter(
            MenuItem.restaurant_id == restaurant.id, MenuItem.id.in_(submitted_ids)
        ).all()}
    else:
        valid_ids = set()

    existing_ids = {s.suggested_item_id for s in
                    MenuItemSuggestion.query.filter_by(menu_item_id=item.id).all()}

    if valid_ids != existing_ids:
        MenuItemSuggestion.query.filter_by(menu_item_id=item.id).delete()
        for sid in valid_ids:
            db.session.add(MenuItemSuggestion(
                restaurant_id=restaurant.id, menu_item_id=item.id, suggested_item_id=sid))
        changes.append("suggestions de vente additionnelle mises à jour")

    if changes:
        log_menu_change(restaurant.id, current_user, "item_edited",
                         f"Plat « {new_name} » modifié : {', '.join(changes)}.")
        db.session.commit()
        flash(f"Plat « {new_name} » mis à jour.", "success")
    else:
        db.session.commit()
    return redirect(url_for("admin_dashboard"))

@app.route("/admin/commandes/stream")
@login_required
def admin_orders_stream():
    restaurant_id = active_restaurant_id()
    q = realtime.subscribe(restaurant_id)

    def event_stream():
        import time
        started = time.time()
        try:
            yield "retry: 3000\n\n"
            # Ferme le flux après 5 minutes : le navigateur se reconnecte
            # automatiquement, ce qui libère les connexions abandonnées
            # (onglets fermés, téléphones en veille).
            while time.time() - started < 300:
                try:
                    message = q.get(timeout=15)
                    yield realtime.format_sse(message)
                except Exception:
                    yield ": ping\n\n"
        finally:
            realtime.unsubscribe(restaurant_id, q)

    return Response(event_stream(), mimetype="text/event-stream", headers={
        "Cache-Control": "no-cache",
        "X-Accel-Buffering": "no",
    })

    db.session.add(restaurant)
    db.session.commit()

    owner = User(restaurant_id=restaurant.id, email=owner_email, role="owner")
    owner.set_password(owner_password)
    db.session.add(owner)
    db.session.commit()

    log_audit("super_admin_restaurant_created", restaurant_id=restaurant.id,
               user_email=current_user.email, details=f"restaurant « {name} » créé par le super-admin")
    flash(f"Restaurant « {name} » créé avec un essai gratuit de {TRIAL_DAYS} jours.", "success")
    return redirect(url_for("super_admin_dashboard"))


@app.route("/super-admin/restaurants/<int:restaurant_id>/payment", methods=["POST"])
@super_admin_required
def super_admin_record_payment(restaurant_id):
    restaurant = Restaurant.query.get_or_404(restaurant_id)

    amount = safe_int(request.form.get("amount"), min_value=0, max_value=100_000_000)
    method = request.form.get("method", "Non précisé")[:30]
    reference = request.form.get("reference", "").strip()[:80]
    months = safe_int(request.form.get("months"), default=1, min_value=1, max_value=36)

    if amount is None:
        flash("Montant invalide.", "error")
        return redirect(url_for("super_admin_dashboard"))

    base_date = restaurant.subscription_active_until
    if not base_date or base_date < datetime.utcnow():
        base_date = datetime.utcnow()
    restaurant.subscription_active_until = base_date + timedelta(days=30 * months)
    restaurant.subscription_plan = request.form.get("plan", restaurant.subscription_plan)[:20]
    restaurant.is_active = True

    db.session.add(SubscriptionPayment(
        restaurant_id=restaurant.id,
        amount=amount,
        method=method,
        reference=reference,
        months_added=months,
        recorded_by=current_user.id,
    ))
    db.session.commit()

    log_audit("super_admin_payment_recorded", restaurant_id=restaurant.id,
               user_email=current_user.email,
               details=f"{amount} FCFA, {months} mois, méthode={method}")
    flash(f"Paiement enregistré : {restaurant.name} est actif jusqu'au "
          f"{restaurant.subscription_active_until.strftime('%d/%m/%Y')}.", "success")
    return redirect(url_for("super_admin_dashboard"))

def _ensure_column(conn, table, column, ddl_type_default):
    cols = [row[1] for row in conn.exec_driver_sql(f"PRAGMA table_info({table})")]
    if column not in cols:
        conn.exec_driver_sql(f"ALTER TABLE {table} ADD COLUMN {column} {ddl_type_default}")


def _ensure_schema_migrations():
    try:
        with db.engine.connect() as conn:
            _ensure_column(conn, "restaurants", "theme", f"VARCHAR(20) DEFAULT '{DEFAULT_THEME}'")
            _ensure_column(conn, "menu_categories", "is_paused", "BOOLEAN DEFAULT 0")
            _ensure_column(conn, "menu_items", "prep_minutes", f"INTEGER DEFAULT {DEFAULT_PREP_MINUTES}")
            _ensure_column(conn, "orders", "comment", "VARCHAR(500)")
            _ensure_column(conn, "orders", "reviewed", "BOOLEAN DEFAULT 0")
            _ensure_column(conn, "orders", "cancelled_by", "VARCHAR(20)")
            _ensure_column(conn, "order_items", "prep_minutes", f"INTEGER DEFAULT {DEFAULT_PREP_MINUTES}")
            _ensure_column(conn, "users", "failed_login_attempts", "INTEGER DEFAULT 0")
            _ensure_column(conn, "users", "locked_until", "DATETIME")
            _ensure_column(conn, "users", "last_login_at", "DATETIME")
            _ensure_column(conn, "users", "security_stamp", "VARCHAR(32) DEFAULT ''")
            _ensure_column(conn, "users", "email_verified", "BOOLEAN DEFAULT 0")
            _ensure_column(conn, "users", "email_verified_at", "DATETIME")
    except Exception:
        pass


def _ensure_security_defaults():
    try:
        users_without_stamp = User.query.filter(
            (User.security_stamp.is_(None)) | (User.security_stamp == "")
        ).all()
        for u in users_without_stamp:
            u.rotate_security_stamp()
        if users_without_stamp:
            db.session.commit()
    except Exception:
        db.session.rollback()


if __name__ == "__main__":
    with app.app_context():
        db.create_all()
        _ensure_schema_migrations()
        _ensure_security_defaults()
    app.run(debug=True, port=5000, host="0.0.0.0", threaded=True)

==================== mailer.py ====================
"""
Envoi d'emails transactionnels (vérification d'adresse, réinitialisation
de mot de passe) via Flask-Mail, avec repli automatique en l'absence de
configuration SMTP (le lien est alors uniquement journalisé, utile pour
tester en local mais à configurer avant tout déploiement réel).
"""
import logging
from flask_mail import Mail, Message

logger = logging.getLogger("menuflash.mailer")
mail = Mail()


def init_mailer(app):
    app.config.setdefault("MAIL_SERVER", "")
    app.config.setdefault("MAIL_PORT", 587)
    app.config.setdefault("MAIL_USE_TLS", True)
    app.config.setdefault("MAIL_USERNAME", "")
    app.config.setdefault("MAIL_PASSWORD", "")
    app.config.setdefault("MAIL_DEFAULT_SENDER", "no-reply@menuflash.local")
    mail.init_app(app)


def is_mail_configured(app):
    return bool(app.config.get("MAIL_SERVER"))


def send_transactional_email(app, to, subject, body_text):
    if is_mail_configured(app):
        try:
            msg = Message(subject=subject, recipients=[to], body=body_text)
            mail.send(msg)
            return True
        except Exception:
            logger.exception("Échec de l'envoi d'email à %s (sujet: %s)", to, subject)
            return False
    else:
        logger.warning(
            "SMTP non configuré — email NON envoyé réellement. "
            "Destinataire: %s | Sujet: %s\n%s",
            to, subject, body_text,
        )
        return False

    "emerald":  {"label": "Émeraude", "accent": "#1F6F4F", "accent_dark": "#164F38", "accent_light": "#DCEFE4"},

class Restaurant(db.Model):
    __tablename__ = "restaurants"

    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(120), nullable=False)
    slug = db.Column(db.String(80), unique=True, nullable=False, index=True)
    phone = db.Column(db.String(30))

    logo_url = db.Column(db.String(300))
    currency = db.Column(db.String(10), default="FCFA")
    language = db.Column(db.String(5), default="fr")
    theme = db.Column(db.String(20), default=DEFAULT_THEME)

                                  order_by="MenuCategory.position")
    qrcodes = db.relationship("QRCode", backref="restaurant", lazy=True,
                               order_by="QRCode.created_at")

    @property
    def theme_colors(self):
        return THEME_PRESETS.get(self.theme, THEME_PRESETS[DEFAULT_THEME])

    @property
    def is_subscription_valid(self):
        if not self.subscription_active_until:
            return False
        return self.subscription_active_until >= datetime.utcnow()

    @property
    def days_remaining(self):
        if not self.subscription_active_until:
            return 0
        delta = self.subscription_active_until - datetime.utcnow()
        return max(delta.days, 0)


class User(UserMixin, db.Model):
    __tablename__ = "users"

    id = db.Column(db.Integer, primary_key=True)
    restaurant_id = db.Column(db.Integer, db.ForeignKey("restaurants.id"), nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(255), nullable=False)
    role = db.Column(db.String(20), default="owner")
    is_super_admin = db.Column(db.Boolean, default=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

    failed_login_attempts = db.Column(db.Integer, default=0, nullable=False)
    locked_until = db.Column(db.DateTime, nullable=True)
    last_login_at = db.Column(db.DateTime, nullable=True)

    security_stamp = db.Column(db.String(32), nullable=False, default=lambda: secrets.token_hex(16))

    email_verified = db.Column(db.Boolean, default=False, nullable=False)
    email_verified_at = db.Column(db.DateTime, nullable=True)

    def set_password(self, password):
        self.password_hash = generate_password_hash(password)
        self.rotate_security_stamp()

    def check_password(self, password):
        return check_password_hash(self.password_hash, password)

    def rotate_security_stamp(self):
        self.security_stamp = secrets.token_hex(16)

    def get_id(self):
        return f"{self.id}:{self.security_stamp}"

    @property
    def is_owner(self):
        return self.role == "owner"

    @property
    def is_locked(self):
        return bool(self.locked_until and self.locked_until > datetime.utcnow())

    @property
    def lock_minutes_remaining(self):
        if not self.is_locked:
            return 0
        delta = self.locked_until - datetime.utcnow()
        return max(int(delta.total_seconds() // 60) + 1, 1)

    def register_failed_login(self):
        self.failed_login_attempts = (self.failed_login_attempts or 0) + 1
        if self.failed_login_attempts >= MAX_FAILED_LOGIN_ATTEMPTS:
            from datetime import timedelta
            self.locked_until = datetime.utcnow() + timedelta(minutes=LOCKOUT_DURATION_MINUTES)

    def register_successful_login(self):
        self.failed_login_attempts = 0
        self.locked_until = None
        self.last_login_at = datetime.utcnow()

"""
Génération du PDF "chevalet de table" imprimable pour un QR code.
Format A6 (105 x 148 mm), pensé pour être imprimé et plié/posé sur une table.
"""
import io
import qrcode
from reportlab.lib.pagesizes import mm
from reportlab.pdfgen import canvas
from reportlab.lib.utils import ImageReader
from reportlab.lib.colors import HexColor

INK = HexColor("#1A1D23")
INK_SOFT = HexColor("#6B6A66")


def make_table_tent_pdf(restaurant_name, qr_target_url, table_label=None, accent_hex="#D98E1F"):
    accent = HexColor(accent_hex)
    page_w, page_h = 105 * mm, 148 * mm
    buf = io.BytesIO()
    c = canvas.Canvas(buf, pagesize=(page_w, page_h))

    c.setFillColor(INK)
    c.rect(0, page_h - 22 * mm, page_w, 22 * mm, fill=1, stroke=0)
    c.setFillColor(HexColor("#FFFFFF"))
    c.setFont("Helvetica-Bold", 16)
    c.drawCentredString(page_w / 2, page_h - 14 * mm, restaurant_name[:28])

    qr = qrcode.QRCode(box_size=10, border=1)
    qr.add_data(qr_target_url)
    qr.make(fit=True)
    img = qr.make_image(fill_color="black", back_color="white")
    qr_buf = io.BytesIO()
    img.save(qr_buf, format="PNG")
    qr_buf.seek(0)

    qr_size = 68 * mm
    qr_x = (page_w - qr_size) / 2
    qr_y = page_h - 22 * mm - qr_size - 10 * mm
    c.drawImage(ImageReader(qr_buf), qr_x, qr_y, width=qr_size, height=qr_size)

    c.setFillColor(INK)
    c.setFont("Helvetica-Bold", 12)
    c.drawCentredString(page_w / 2, qr_y - 8 * mm, "Scannez pour voir le menu")

    if table_label:
        c.setFont("Helvetica", 10)
        c.setFillColor(accent)
        c.drawCentredString(page_w / 2, qr_y - 14 * mm, table_label)

    c.setFont("Helvetica", 7)
    c.setFillColor(INK_SOFT)
    c.drawCentredString(page_w / 2, 6 * mm, "Propulsé par MenuFlash Mini")

    c.showPage()
    c.save()
    buf.seek(0)
    return buf

# MenuFlash Mini — Starter technique

MVP de menu par QR code pour restaurants, avec identité visuelle unifiée,
protections de sécurité renforcées, suggestions de vente additionnelle,
mode multi-restaurant, et désormais une **navigation réorganisée** et une
**identité de marque propre à chaque restaurant**.

## Démarrage local (Windows / PowerShell)

```powershell
py -3.13 -m venv venv
venv\Scripts\activate
pip install -r requirements.txt
$env:FLASK_SECRET_KEY = "change-moi"
python seed.py
python app.py
```

- **Page de présentation** : http://127.0.0.1:5000/
- **Inscription en libre-service** : http://127.0.0.1:5000/inscription
- Menu public de démo : http://127.0.0.1:5000/m/chez-awa
- Connexion gestionnaire : `awa@example.com` / `motdepasse123`
- Connexion cuisinier (accès limité) : `cuisine@example.com` / `cuisine1234`
- Panel super-admin : `admin@menuflash.com` / `SuperAdmin123!`

⚠️ **Important** : `requirements.txt` épingle `Flask==3.0.3` et
`Werkzeug==3.0.6`. Vérifie toujours avec
`pip show flask werkzeug flask-login` avant de signaler un bug étrange
(une version plus récente de Werkzeug peut provoquer des erreurs de
session difficiles à diagnostiquer avec Flask-Login 0.6.3).

## Nouveau : navigation réorganisée en menus déroulants

La barre de navigation du back-office (qui alignait une dizaine de
boutons et finissait par se replier sur deux lignes) est maintenant
regroupée en :
- **2 liens directs** pour les actions les plus fréquentes : **Menu**
  (gestion des plats) et **Commandes** (avec le badge du nombre de
  commandes en cours, désormais visible sur toutes les pages).
- **3 menus déroulants thématiques** :
  - **Boutique** : Voir le menu public, Codes QR, Apparence & logo.
  - **Gestion** : Statistiques, Équipe, Historique.
  - **Compte** : Journal de sécurité, Mot de passe, Déconnexion.

Cette barre de navigation est désormais centralisée dans un seul fichier
(`templates/_admin_navbar.html`), inclus par les 8 pages du back-office —
une évolution future n'aura qu'un seul endroit à modifier, au lieu de
risquer que les pages divergent entre elles.

## Nouveau : identité visuelle propre à chaque restaurant

Une fois connecté à son tableau de bord, un restaurateur voit désormais
**le nom et le logo de SON restaurant** à la place de la marque
« MenuFlash Mini » dans la barre de navigation — exactement comme s'il
s'agissait de son propre logiciel.

### Ajouter un logo
Depuis **Boutique > Apparence & logo**, un formulaire permet de
téléverser une image (JPG, PNG ou WEBP). Particularité importante : si
l'image a un fond transparent (PNG), cette transparence est **préservée**
dans le fichier final — un logo ne se retrouve jamais avec un carré blanc
disgracieux sur la barre de navigation sombre. Sans transparence, le
fichier est converti en WEBP (plus léger). Le logo apparaît ensuite :
- dans la barre de navigation de tout le back-office,
- sur le menu public que voient les clients (déjà géré par le composant
  existant, aucun changement nécessaire côté menu public),
- peut être supprimé à tout moment (la marque « MenuFlash Mini » reprend
  alors sa place).

### Ce qui garde la marque MenuFlash Mini
- Le **panel super-admin** (qui gère l'ensemble de la plateforme, pas un
  restaurant précis) conserve sa propre barre de navigation inchangée.
- Les pages **avant connexion** (page de présentation, connexion,
  inscription, mot de passe oublié) : logique, puisqu'aucun restaurant
  n'est encore identifié à ce stade.

Ces deux changements ont été vérifiés par une trentaine de tests
automatisés : présence correcte des 3 menus déroulants et des 2 liens
directs sur les 8 pages concernées, affichage du nom/logo du restaurant
(et non de la marque de la plateforme) dans chaque cas, préservation
réelle de la transparence PNG sur le fichier enregistré sur disque,
redimensionnement correct, gestion propre d'un fichier corrompu (sans
erreur serveur), suppression du logo, bascule correcte de l'icône
générique, badge de commandes désormais visible sur toutes les pages
(plus seulement le tableau de bord et la page Commandes), et isolation
confirmée pour le compte super-admin (marque plateforme conservée) et le
compte cuisinier (navigation strictement minimale, badge "Cuisine").

## Fonctionnalités (rappel)

- Page de présentation + inscription en libre-service.
- Commande en ligne, notifications cuisine instantanées, annulation de
  commande par le client.
- Suggestions de vente additionnelle, mode multi-restaurant.
- Thèmes en direct, statistiques de ventes, comptes d'équipe à rôles,
  avis clients par plat, interface FR/EN, mode rush, historique des
  modifications.
- Sécurité : CSRF, limitation de débit, en-têtes HTTP (CSP, HSTS),

  réinitialisation sécurisée, journal d'audit.

Les emails (vérification, réinitialisation) nécessitent un serveur SMTP

"""
Diffusion d'événements en temps réel (Server-Sent Events), par restaurant.
"""
import json
import queue
import threading

_subscribers = {}
_lock = threading.Lock()


def subscribe(restaurant_id):
    q = queue.Queue()
    with _lock:
        _subscribers.setdefault(restaurant_id, []).append(q)
    return q


def unsubscribe(restaurant_id, q):
    with _lock:
        subs = _subscribers.get(restaurant_id)
        if subs and q in subs:
            subs.remove(q)
            if not subs:
                _subscribers.pop(restaurant_id, None)


def publish(restaurant_id, event_type, payload):
    message = json.dumps({"type": event_type, "data": payload})
    with _lock:
        subs = list(_subscribers.get(restaurant_id, []))
    for q in subs:
        q.put(message)


def format_sse(message):
    return f"data: {message}\n\n"

  (function installCsrfFetchGuard() {
    var nativeFetch = window.fetch;

      var meta = document.querySelector('meta[name="csrf-token"]');

      if (method !== "GET" && method !== "HEAD") {

        } else {

        }
      }

  function initOrderCartAndMyOrders(requestCancelConfirmation) {
    var form = document.getElementById("order-form");

    var storageKey = "mf_orders_" + slug;

    var cartBar = document.getElementById("cart-bar");

    var totalEl = document.getElementById("cart-total");
    var submitBtn = document.getElementById("cart-submit");
    var submitLabelEl = document.getElementById("cart-submit-label");

    var myOrdersList = document.getElementById("my-orders-list");
    var myOrdersEmpty = document.getElementById("my-orders-empty");
    var myOrdersBadge = document.getElementById("my-orders-badge");

    var addmodeBanner = document.getElementById("addmode-banner");
    var addmodeOrderRef = document.getElementById("addmode-order-ref");
    var addmodeCancelBtn = document.getElementById("addmode-cancel");

    var addModeOrderId = null;

    var suggestionsDict = loadJsonScript("mf-suggestions-data") || {};
    var nudgedItemIds = new Set();

    function getSuggestionsFor(itemId) {
      return suggestionsDict[String(itemId)] || [];
    }

    function computeUpsellSuggestions(cartItemIds) {
      var inCart = new Set(cartItemIds);
      var seen = new Set();
      var result = [];
      cartItemIds.forEach(function (id) {
        getSuggestionsFor(id).forEach(function (sug) {
          if (!inCart.has(sug.id) && !seen.has(sug.id)) {
            seen.add(sug.id);
            result.push(sug);
          }
        });
      });
      return result.slice(0, 4);
    }

    function getStoredIds() {
      try {
        var raw = localStorage.getItem(storageKey);
        var ids = raw ? JSON.parse(raw) : [];
        return Array.isArray(ids) ? ids : [];
      } catch (e) { return []; }
    }
    function addStoredId(id) {
      var ids = getStoredIds();
      if (ids.indexOf(id) === -1) {
        ids.push(id);
        try { localStorage.setItem(storageKey, JSON.stringify(ids)); } catch (e) {}
      }
    }

    function cartArticleLabel(count) {
      var key = count > 1 ? "cart_article_other" : "cart_article_one";
      return translateKey(key) || (count > 1 ? "articles" : "article");
    }

    function readCart() {
      var count = 0, total = 0, lines = [], ids = [];
      form.querySelectorAll(".mf-qty-group").forEach(function (group) {
        var qty = parseInt(group.querySelector(".mf-qty-value").textContent, 10) || 0;
        var price = parseInt(group.dataset.price, 10) || 0;
        var name = group.dataset.name || "";
        var itemId = parseInt(group.dataset.itemId, 10);
        if (qty > 0) {
          count += qty;
          total += qty * price;
          lines.push({ name: name, qty: qty, price: price });
          if (!isNaN(itemId)) ids.push(itemId);
        }
      });
      return { count: count, total: total, lines: lines, ids: ids };
    }

    function updateCartUI() {
      var cart = readCart();
      if (countEl) countEl.textContent = cart.count + " " + cartArticleLabel(cart.count);
      if (totalEl) totalEl.textContent = formatNumber(cart.total) + " " + currency;
      if (submitBtn) submitBtn.disabled = cart.count === 0;
      if (cartBar) cartBar.classList.toggle("is-visible", cart.count > 0);
      return cart;
    }

    function resetCart() {
      form.querySelectorAll(".mf-qty-group").forEach(function (group) {
        group.querySelector(".mf-qty-value").textContent = "0";
        group.querySelector("input[type=hidden]").value = "0";
      });
      updateCartUI();
    }

    function incrementItemQty(itemId) {
      var group = form.querySelector('.mf-qty-group[data-item-id="' + itemId + '"]');
      if (!group) return;
      var valueEl = group.querySelector(".mf-qty-value");
      var hiddenInput = group.querySelector("input[type=hidden]");
      var newQty = Math.min((parseInt(hiddenInput.value, 10) || 0) + 1, 50);
      hiddenInput.value = newQty;
      valueEl.textContent = newQty;
      valueEl.classList.remove("mf-pop");
      void valueEl.offsetWidth;
      valueEl.classList.add("mf-pop");
    }

    form.querySelectorAll(".mf-qty-group").forEach(function (group) {
      var valueEl = group.querySelector(".mf-qty-value");
      var hiddenInput = group.querySelector("input[type=hidden]");
      var minusBtn = group.querySelector(".mf-qty-minus");
      var plusBtn = group.querySelector(".mf-qty-plus");
      var itemId = parseInt(group.dataset.itemId, 10);

      function setQty(q) {
        var previous = parseInt(hiddenInput.value, 10) || 0;
        q = Math.max(0, Math.min(50, q));
        valueEl.textContent = q;
        hiddenInput.value = q;
        valueEl.classList.remove("mf-pop");
        void valueEl.offsetWidth;
        valueEl.classList.add("mf-pop");
        updateCartUI();

        if (previous === 0 && q > 0 && !isNaN(itemId) && !nudgedItemIds.has(itemId)) {
          var cart = readCart();
          var sugg = computeUpsellSuggestions(cart.ids).filter(function (s) {
            return getSuggestionsFor(itemId).some(function (own) { return own.id === s.id; });
          });
          if (sugg.length > 0) {
            nudgedItemIds.add(itemId);
            var prefix = translateKey("upsell_nudge_prefix") || "💡 Nos clients ajoutent souvent :";
            var names = sugg.map(function (s) { return s.name; }).join(", ");
            showToast(prefix + " " + names, false);
          }
        }
      }

      minusBtn.addEventListener("click", function () {
        setQty((parseInt(hiddenInput.value, 10) || 0) - 1);
      });
      plusBtn.addEventListener("click", function () {
        setQty((parseInt(hiddenInput.value, 10) || 0) + 1);
      });
    });

          headers: { "X-Requested-With": "fetch" },

    function updateBadge(activeCount) {
      if (!myOrdersBadge) return;
      if (activeCount > 0) {
        myOrdersBadge.textContent = activeCount;
        myOrdersBadge.classList.remove("d-none");
      } else {
        myOrdersBadge.classList.add("d-none");
      }
    }

    function refreshMyOrders() {
      if (!myOrdersList) return;
      var ids = getStoredIds();
      if (ids.length === 0) {
        if (myOrdersEmpty) myOrdersEmpty.style.display = "";
        updateBadge(0);
        return;
      }
      fetch("/m/" + slug + "/commandes/etats.json?ids=" + ids.join(","))
        .then(function (r) { return r.json(); })
        .then(function (data) {
          var orders = data.orders || [];
          if (orders.length === 0) {
            if (myOrdersEmpty) myOrdersEmpty.style.display = "";
            myOrdersList.innerHTML = "";
            updateBadge(0);
            return;
          }
          if (myOrdersEmpty) myOrdersEmpty.style.display = "none";
          myOrdersList.innerHTML = orders.slice().reverse().map(renderMyOrder).join("");

          var activeCount = orders.filter(function (o) {
            return o.status !== "servie" && o.status !== "annulee";
          }).length;
          updateBadge(activeCount);
        })
        .catch(function () {});
    }

    if (myOrdersList) {
      myOrdersList.addEventListener("click", function (e) {
        var addBtn = e.target.closest(".mf-addto-btn");
        if (addBtn) {
          enterAddMode(parseInt(addBtn.dataset.orderId, 10), addBtn.dataset.tableLabel);
          return;
        }
        var cancelBtn = e.target.closest(".mf-cancel-order-trigger");
        if (cancelBtn && requestCancelConfirmation) {
          var orderId = parseInt(cancelBtn.dataset.orderId, 10);
          requestCancelConfirmation(function (finish) {
            cancelOrder(orderId, finish);
          });
        }
      });
    }

    updateCartUI();
    refreshMyOrders();
    setInterval(refreshMyOrders, 6000);
  }

  function initCategoryNav() {
    var navLinks = document.querySelectorAll(".mf-nav-pills .nav-link");
    if (!navLinks.length) return;
    var sections = Array.prototype.map.call(navLinks, function (link) {
      return document.querySelector(link.getAttribute("href"));
    });

    function onScroll() {
      var scrollPos = window.scrollY + 120;
      var activeIndex = 0;
      sections.forEach(function (sec, i) {
        if (sec && sec.offsetTop <= scrollPos) activeIndex = i;
      });
      navLinks.forEach(function (link, i) {
        link.classList.toggle("active", i === activeIndex);
      });
    }
    window.addEventListener("scroll", onScroll, { passive: true });
    onScroll();
  }

  function initOrderStatusPolling(requestCancelConfirmation) {
    var root = document.getElementById("order-status-root");
    if (!root) return;
    var pollUrl = root.dataset.pollUrl;
    var cancelUrl = root.dataset.cancelUrl;
    var stepOrder = ["nouvelle", "en_preparation", "prete", "servie"];
    var badge = document.getElementById("status-badge");
    var progressBar = document.getElementById("status-progress-bar");
    var stepLabels = document.querySelectorAll(".mf-step-labels span");
    var cancelBtn = document.getElementById("cancel-order-btn");

    function applyStatus(status, label) {
      var translated = (window.MFTranslate && window.MFTranslate("status_" + status)) || label;
      badge.textContent = translated;
      badge.setAttribute("data-i18n", "status_" + status);
      badge.className = "mf-status-badge badge mf-status-" + status;
      var idx = stepOrder.indexOf(status);
      var pct = idx >= 0 ? ((idx + 1) / stepOrder.length) * 100 : 100;
      if (progressBar) progressBar.style.width = pct + "%";
      stepLabels.forEach(function (el, i) {
        el.classList.toggle("active", i <= idx);
      });
    }

    if (cancelBtn && cancelUrl && requestCancelConfirmation) {
      cancelBtn.addEventListener("click", function () {
        requestCancelConfirmation(function (finish) {
          fetch(cancelUrl, { method: "POST", headers: { "X-Requested-With": "fetch" } })
            .then(function (r) { return r.json(); })
            .then(function (data) {
              finish();
              if (data.ok) {
                showToast(translateKey("cancel_order_success") || "Commande annulée.", false);
                setTimeout(function () { window.location.reload(); }, 800);
              } else if (data.locked) {
                showToast(translateKey("cancel_order_locked") || "Trop tard : cette commande est déjà en préparation.", true);
                setTimeout(function () { window.location.reload(); }, 800);
              } else {
                showToast(data.error || "Impossible d'annuler cette commande.", true);
              }
            })
            .catch(function () {
              finish();
              showToast("Impossible d'annuler la commande. Vérifiez votre connexion.", true);
            });
        });
      });
    }

    var currentStatus = root.dataset.status;
    applyStatus(currentStatus, root.dataset.statusLabel);

    if (currentStatus === "servie" || currentStatus === "annulee") return;

    var interval = setInterval(function () {
      fetch(pollUrl).then(function (r) { return r.json(); }).then(function (data) {
        if (data.status !== currentStatus) {
          currentStatus = data.status;
          applyStatus(data.status, data.status_label);
          badge.classList.add("mf-pulse");
          if (currentStatus === "servie" || currentStatus === "en_preparation") {
            setTimeout(function () { window.location.reload(); }, 1200);
          }
        }
        if (currentStatus === "servie" || currentStatus === "annulee") {
          clearInterval(interval);
        }
      }).catch(function () {});
    }, 5000);
  }

    function renderCard(order, nextLabel, isNew) {
      var itemsHtml = order.items.map(function (it) {
        return "<li>" + it.quantity + " × " + it.name + "</li>";
      }).join("");
      var noteHtml = order.notes ? '<p class="mf-order-note mb-2">📝 ' + order.notes + "</p>" : "";
      return (
        '<div class="card mf-order-card-k mb-2 ' + (isNew ? "mf-animate-in mf-pulse" : "") + '" data-order-id="' + order.id + '">' +
          '<div class="card-body p-3">' +
            '<div class="d-flex justify-content-between align-items-center mb-2">' +
              '<strong>' + order.table_label + '</strong>' +
              '<span class="mf-elapsed">' + order.elapsed_minutes + ' min</span>' +
            '</div>' +
            '<ul class="mb-2 ps-3 small">' + itemsHtml + '</ul>' +
            noteHtml +
            '<p class="fw-bold mb-2" style="color: var(--mf-accent-dark);">' +
              order.total.toLocaleString("fr-FR") + ' ' + currency +
            '</p>' +
            '<div class="d-flex gap-2">' +
              '<button class="btn btn-outline-secondary btn-sm flex-grow-1 mf-advance-btn" data-order-id="' + order.id + '">' + nextLabel + '</button>' +
              '<button class="btn btn-outline-danger btn-sm mf-cancel-btn" data-order-id="' + order.id + '"><i class="bi bi-x-lg"></i></button>' +
            '</div>' +
          '</div>' +
        '</div>'
      );
    }

    function attachActions(container) {
      container.querySelectorAll(".mf-advance-btn").forEach(function (btn) {
        btn.addEventListener("click", function () {
          fetch("/admin/commandes/" + btn.dataset.orderId + "/avancer", {
            method: "POST", headers: { "X-Requested-With": "fetch" }
          }).then(refresh);
        });
      });
      container.querySelectorAll(".mf-cancel-btn").forEach(function (btn) {
        btn.addEventListener("click", function () {
          fetch("/admin/commandes/" + btn.dataset.orderId + "/annuler", {
            method: "POST", headers: { "X-Requested-With": "fetch" }
          }).then(refresh);
        });
      });
    }

    function refresh() {
      fetch(feedUrl).then(function (r) { return r.json(); }).then(function (data) {
        ["nouvelle", "en_preparation", "prete"].forEach(function (status) {
          var col = document.getElementById("col-" + status);
          if (!col) return;
          var orders = data.orders.filter(function (o) { return o.status === status; });
          var countEl = document.getElementById("count-" + status);
          if (countEl) countEl.textContent = orders.length;

          if (orders.length === 0) {
            col.innerHTML = '<p class="text-muted small">Aucune commande.</p>';
            return;
          }
          col.innerHTML = orders.map(function (o) {
            var isNew = !knownIds.has(String(o.id));
            return renderCard(o, columnLabels[status].next, isNew);
          }).join("");
        });

        knownIds = new Set(data.orders.map(function (o) { return String(o.id); }));
        attachActions(root);
      }).catch(function () {});
    }

    function refreshWithCancelAnimation(orderId) {
      var card = root.querySelector('[data-order-id="' + orderId + '"]');
      if (card) {
        card.classList.add("mf-cancelling");
        setTimeout(refresh, 380);
      } else {
        refresh();
      }
    }

    attachActions(root);
    refresh();
    setInterval(refresh, 20000);

    kanbanRefreshFn = refresh;
    window.MFKanbanRefreshWithCancelAnimation = refreshWithCancelAnimation;
  }

          setTimeout(function () { btn.innerHTML = original; }, 1500);
        });
      });
    });
  }

    /* ------------------------------------------------------------------
     SURVEILLANCE DES COMMANDES sur toutes les pages du back-office (hors
     page Commandes, qui a son propre temps réel SSE). Interroge un point
     de contrôle léger toutes les 8 s (30 s si l'onglet est en arrière-
     plan) et, à chaque nouvelle commande : son + notification + mise à
     jour du badge et du bandeau, sans recharger la page.

  function escapeHtml(text) {
    return String(text == null ? "" : text)
      .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;").replace(/'/g, "&#39;");
  }

  function initGlobalOrderWatcher() {
    var el = document.getElementById("mf-order-watcher");
    if (!el) return;

    var url = el.dataset.url;
    var storageKey = "mf_last_order_id";
    var badge = document.getElementById("nav-orders-badge");
    var banner = document.getElementById("mf-orders-banner");
    var bannerText = document.getElementById("mf-orders-banner-text");
    var timer = null;
    var lastId = null;

    try {
      var stored = sessionStorage.getItem(storageKey);
      if (stored !== null) lastId = parseInt(stored, 10);
    } catch (e) {}

    function render(count) {
      if (badge) {
        badge.textContent = count;
        badge.classList.toggle("d-none", count === 0);
      }
      if (banner && bannerText) {
        bannerText.textContent = "Vous avez " + count + " commande" + (count > 1 ? "s" : "") + " en cours.";
        banner.classList.toggle("d-none", count === 0);
      }
    }

    function poll() {
      fetch(url, { cache: "no-store", headers: { "X-Requested-With": "fetch" } })
        .then(function (r) { return r.ok ? r.json() : null; })
        .then(function (data) {
          if (!data) return;
          render(data.count);

          if (lastId !== null && data.latest_id > lastId) {
            playNotificationSound();
            showToast(
              "🆕 Nouvelle commande · " + escapeHtml(data.latest_table) +
              ' — <a href="/admin/commandes" class="fw-bold">Voir</a>',
              false
            );
            if (badge) {
              badge.classList.remove("mf-pop");
              void badge.offsetWidth;
              badge.classList.add("mf-pop");
            }
          }

          lastId = data.latest_id;
          try { sessionStorage.setItem(storageKey, String(lastId)); } catch (e) {}
        })
        .catch(function () { /* réseau indisponible : on réessaiera */ });
    }

    function schedule() {
      clearInterval(timer);
      timer = setInterval(poll, document.hidden ? 30000 : 8000);
    }

    document.addEventListener("visibilitychange", function () {
      if (!document.hidden) poll();
      schedule();
    });

    poll();
    schedule();
    initGlobalOrderWatcher();
    initThemeLiveSync();
  }

  function initThemeLiveSync() {
    var el = document.getElementById("theme-sync-root") || document.getElementById("public-theme-sync-root");
    if (!el || typeof EventSource === "undefined") return;

    var streamUrl = el.dataset.streamUrl;
    var isAdminChannel = el.id === "theme-sync-root";
    var liveIndicator = document.getElementById("live-indicator");
    var reconnectDelay = 2000;

    function setLiveStatus(isLive) {
      if (!liveIndicator) return;
      liveIndicator.classList.toggle("mf-live-off", !isLive);
      liveIndicator.innerHTML = isLive
        ? '<span class="mf-live-dot"></span> En direct'
        : '<span class="mf-live-dot"></span> Reconnexion…';
    }

    function connect() {
      var source = new EventSource(streamUrl);

      source.onopen = function () {
        setLiveStatus(true);
        reconnectDelay = 2000;
      };

      source.onmessage = function (event) {
        var payload;
        try { payload = JSON.parse(event.data); } catch (e) { return; }

        if (payload.type === "theme_changed") {
          applyThemeVars(payload.data);
          if (isAdminChannel) showToast("🎨 Le thème a été mis à jour.", false);
          return;
        }

        if (!isAdminChannel) return;

        var tableInfo = (payload.data && payload.data.table_label) ? " · " + payload.data.table_label : "";

        if (payload.type === "new_order") {
          playNotificationSound();
          showToast("🆕 Nouvelle commande reçue" + tableInfo + " !", false);
          if (kanbanRefreshFn) kanbanRefreshFn();
        } else if (payload.type === "order_updated") {
          playNotificationSound();
          showToast("✏️ Une commande a été mise à jour" + tableInfo + ".", false);
          if (kanbanRefreshFn) kanbanRefreshFn();
        } else if (payload.type === "order_cancelled") {
          showToast("🚫 Commande annulée par le client" + tableInfo + ".", false);
          var orderId = payload.data && payload.data.id;
          if (orderId && window.MFKanbanRefreshWithCancelAnimation) {
            window.MFKanbanRefreshWithCancelAnimation(orderId);
          } else if (kanbanRefreshFn) {
            kanbanRefreshFn();
          }
        } else if (kanbanRefreshFn) {
          kanbanRefreshFn();
        }
      };

      source.onerror = function () {
        setLiveStatus(false);
        source.close();
        setTimeout(connect, reconnectDelay);
        reconnectDelay = Math.min(reconnectDelay * 1.5, 15000);
      };
    }

    connect();
  }

  document.addEventListener("DOMContentLoaded", function () {
    var requestCancelConfirmation = initCancelOrderModal();

    initI18n();
    initToasts();
    initConfirmModal();
    initRestaurantSwitcherModal();
    initOrderCartAndMyOrders(requestCancelConfirmation);
    initCategoryNav();
    initOrderStatusPolling(requestCancelConfirmation);
    initReviewForm();
    initSortable();
    initKitchenPolling();
    initCopyButtons();
    initPaymentModal();
    initThemeLiveSync();
  });
})();

"""
Calculs du tableau de bord "Statistiques" : chiffre d'affaires par période,
plats les plus vendus, répartition des commandes par heure.
"""
from collections import Counter
from datetime import datetime, timedelta

from models import db, Order, OrderItem


def _period_bounds(now=None):
    now = now or datetime.utcnow()
    today_start = datetime(now.year, now.month, now.day)
    week_start = today_start - timedelta(days=today_start.weekday())
    month_start = datetime(now.year, now.month, 1)
    return today_start, week_start, month_start


def _period_stats(restaurant_id, start):
    orders = Order.query.filter(
        Order.restaurant_id == restaurant_id,
        Order.status != "annulee",
        Order.created_at >= start,
    ).all()
    revenue = sum(o.total for o in orders)
    count = len(orders)
    avg = (revenue / count) if count else 0
    return {"revenue": revenue, "count": count, "avg": round(avg)}


def top_selling_items(restaurant_id, days=30, limit=5):
    since = datetime.utcnow() - timedelta(days=days)
    rows = (
        db.session.query(OrderItem.item_name, db.func.sum(OrderItem.quantity))
        .join(Order, Order.id == OrderItem.order_id)
        .filter(
            Order.restaurant_id == restaurant_id,
            Order.status != "annulee",
            Order.created_at >= since,
        )
        .group_by(OrderItem.item_name)
        .order_by(db.func.sum(OrderItem.quantity).desc())
        .limit(limit)
        .all()
    )
    return [{"name": name, "quantity": int(qty)} for name, qty in rows]


def peak_hours(restaurant_id, days=30):
    since = datetime.utcnow() - timedelta(days=days)
    orders = Order.query.filter(
        Order.restaurant_id == restaurant_id,
        Order.status != "annulee",
        Order.created_at >= since,
    ).all()
    counter = Counter(o.created_at.hour for o in orders)
    return [{"hour": h, "count": counter.get(h, 0)} for h in range(24)]


def build_dashboard_stats(restaurant_id, now=None):
    today_start, week_start, month_start = _period_bounds(now)
    return {
        "today": _period_stats(restaurant_id, today_start),
        "week": _period_stats(restaurant_id, week_start),
        "month": _period_stats(restaurant_id, month_start),
        "top_items": top_selling_items(restaurant_id),
        "peak_hours": peak_hours(restaurant_id),
    }

<script>
document.addEventListener("DOMContentLoaded", function () {
  var saveUrl = "{{ url_for('admin_appearance_save') }}";

  document.querySelectorAll(".mf-theme-radio").forEach(function (radio) {
    radio.addEventListener("change", function () {
      var card = radio.closest("label").querySelector(".mf-theme-card");

      document.querySelectorAll(".mf-theme-card").forEach(function (c) {
        c.classList.remove("selected");
      });
      card.classList.add("selected");

      var colors = {
        accent: card.dataset.accent,
        accent_dark: card.dataset.accentDark,
        accent_light: card.dataset.accentLight
      };
      if (window.MFApplyTheme) window.MFApplyTheme(colors);

      var formData = new FormData();
      formData.append("theme", radio.value);

      fetch(saveUrl, {
        method: "POST",
        headers: { "X-Requested-With": "fetch" },
        body: formData
      })
        .then(function (r) { return r.json(); })
        .then(function (data) {
          if (data.ok && window.MFShowToast) {
            window.MFShowToast("Thème « " + card.querySelector("strong").textContent + " » appliqué.", false);
          } else if (!data.ok && window.MFShowToast) {
            window.MFShowToast(data.error || "Impossible d'appliquer ce thème.", true);
          }
        })
        .catch(function () {
          if (window.MFShowToast) window.MFShowToast("Impossible d'enregistrer le thème. Vérifiez votre connexion.", true);
        });
    });
  });
});
</script>

document.addEventListener("DOMContentLoaded", function () {
  var canvas = document.getElementById("peak-hours-chart");
  if (!canvas || typeof Chart === "undefined") return;

  var hoursData = JSON.parse(canvas.dataset.hours || "[]");
  var labels = hoursData.map(function (h) { return h.hour + "h"; });
  var counts = hoursData.map(function (h) { return h.count; });
  var accentColor = getComputedStyle(document.documentElement).getPropertyValue("--mf-accent").trim() || "#D98E1F";

  window.MFPeakHoursChart = new Chart(canvas.getContext("2d"), {
    type: "bar",
    data: {
      labels: labels,
      datasets: [{
        data: counts,
        backgroundColor: accentColor,
        borderRadius: 4,
        maxBarThickness: 18
      }]
    },
    options: {
      plugins: { legend: { display: false } },
      scales: {
        y: { beginAtZero: true, ticks: { precision: 0 } },
        x: { grid: { display: false } }
      }
    }
  });
});

  <script>
    (function () {
      var nameInput = document.getElementById("restaurant-name-input");
      var slugInput = document.getElementById("slug-input");
      var feedback = document.getElementById("slug-feedback");
      var slugEditedManually = false;
      var checkTimer = null;

      function slugifyClientSide(value) {
        return value
          .normalize("NFD").replace(/[\u0300-\u036f]/g, "")
          .toLowerCase().trim()
          .replace(/[^a-z0-9]+/g, "-")
          .replace(/-+/g, "-")
          .replace(/^-|-$/g, "");
      }

      function checkSlug(value) {
        if (!value) {
          feedback.textContent = "Laissez vide pour générer automatiquement à partir du nom.";
          feedback.className = "form-text";
          return;
        }
        fetch("/inscription/verifier-slug?slug=" + encodeURIComponent(value))
          .then(function (r) { return r.json(); })
          .then(function (data) {
            if (data.available) {
              feedback.textContent = "✓ Disponible : menuflash.app/m/" + data.slug;
              feedback.className = "form-text text-success";
            } else {
              feedback.textContent = "Déjà pris — un identifiant proche vous sera proposé automatiquement.";
              feedback.className = "form-text text-warning";
            }
          })
          .catch(function () {});
      }

      nameInput.addEventListener("input", function () {
        if (!slugEditedManually) {
          slugInput.value = slugifyClientSide(nameInput.value);
          clearTimeout(checkTimer);
          checkTimer = setTimeout(function () { checkSlug(slugInput.value); }, 300);
        }
      });

      slugInput.addEventListener("input", function () {
        slugEditedManually = true;
        slugInput.value = slugifyClientSide(slugInput.value);
        clearTimeout(checkTimer);
        checkTimer = setTimeout(function () { checkSlug(slugInput.value); }, 300);
      });

      if (slugInput.value) checkSlug(slugInput.value);
    })();
  </script>
